def download_file(url, dest_path):
    """Download a file with progress reporting"""
    print(f"\nDownloading: {os.path.basename(dest_path)}")
    print(f"URL: {url}")
    
    def progress_hook(count, block_size, total_size):
        percent = int(count * block_size * 100 / total_size)
        downloaded = count * block_size / (1024 * 1024)  # MB
        total = total_size / (1024 * 1024)  # MB
        print(f"\rProgress: {percent}% ({downloaded:.1f}/{total:.1f} MB)", end='')
    
    try:
        urllib.request.urlretrieve(url, dest_path, progress_hook)
        print(f"\n✓ Successfully downloaded to: {dest_path}")
        return True
    except Exception as e:
        print(f"\n✗ Error downloading: {e}")
        return False

def main():
    print("=" * 70)
    print("Wan 2.2 Model Downloader for ComfyUI")
    print("=" * 70)
    
    # Create directories and download files
    success_count = 0
    fail_count = 0
    
    for model in models:
        # Create directory path
        dir_path = os.path.join(BASE_DIR, model["category"])
        os.makedirs(dir_path, exist_ok=True)
        print(f"\n📁 Directory: {dir_path}")
        
        # Full file path
        file_path = os.path.join(dir_path, model["filename"])
        
        # Check if file already exists
        if os.path.exists(file_path):
            file_size = os.path.getsize(file_path) / (1024 * 1024)  # MB
            print(f"⚠ File already exists ({file_size:.1f} MB): {model['filename']}")
            user_input = input("Skip download? (y/n): ").lower()
            if user_input == 'y':
                print("Skipped.")
                success_count += 1
                continue
        
        # Download the file
        if download_file(model["url"], file_path):
            success_count += 1
        else:
            fail_count += 1
    
    # Summary
    print("\n" + "=" * 70)
    print("Download Summary")
    print("=" * 70)
    print(f"✓ Successfully downloaded: {success_count}")
    print(f"✗ Failed: {fail_count}")
    print(f"Total files: {len(models)}")
    print("\nAll models are ready for ComfyUI!")

if __name__ == "__main__":
    main()
